1. Normal input/output
	cat < infile.txt > outfile.txt
	==================================
	cmd->args     = ["cat", NULL]
	cmd->infile   = "infile.txt"
	cmd->outfile  = "outfile.txt"
	cmd->append   = 0
	cmd->heredoc  = NULL

2. Append (>>)
	echo hello >> log.txt
	==================================
	cmd->args     = ["echo", "hello", NULL]
	cmd->infile   = NULL
	cmd->outfile  = "log.txt"
	cmd->append   = 1              // tells executor to open with O_APPEND
	cmd->heredoc  = NULL
3. Heredoc (<<)
	cat << EOF
	==================================
	hello
	world
	EOF
	cmd->args     = ["cat", NULL]
	cmd->infile   = NULL
	cmd->outfile  = NULL
	cmd->append   = 0
	cmd->heredoc  = "EOF"         // readline until user types "EOF"

4. Heredoc + append together
	cat << END >> log.txt
	line one
	line two
	END
	==================================
	cmd->args     = ["cat", NULL]
	cmd->infile   = NULL
	cmd->outfile  = "log.txt"
	cmd->append   = 1             // append to log.txt
	cmd->heredoc  = "END"         // read stdin until "END"

5. Full pipeline with mixed redirections
	grep foo < input.txt | sort >> result.txt
	==================================
	// cmd1
	cmd1->args     = ["grep", "foo", NULL]
	cmd1->infile   = "input.txt"
	cmd1->outfile  = NULL          // output goes to pipe, not a file
	cmd1->append   = 0
	cmd1->heredoc  = NULL
	cmd1->next     = cmd2

	// cmd2
	cmd2->args     = ["sort", NULL]
	cmd2->infile   = NULL          // input comes from pipe, not a file
	cmd2->outfile  = "result.txt"
	cmd2->append   = 1
	cmd2->heredoc  = NULL
	cmd2->next     = NULL


RULE of EXECUTOR
	- if heredoc is set → use it as stdin (ignore infile)
	- if infile is set → open it and dup2 to stdin
	- if outfile is set and append == 1 → open with O_WRONLY | O_CREAT | O_APPEND
	- if outfile is set and append == 0 → open with O_WRONLY | O_CREAT | O_TRUNC
	pipe between cmd->next nodes is always handled separately by your pipe loop
